## 程式碼實作

### 1. `etl_weather.py` (HW10-1 ~ HW10-3)

```python
import os
import json
import sqlite3
import requests
import pandas as pd

# ==========================================
# 設定區：請填入您的中央氣象署 CWA API 授權碼
# ==========================================
CWA_API_KEY = os.getenv("CWA_API_KEY", "YOUR_CWA_API_KEY_HERE")
API_URL = "[https://opendata.cwa.gov.tw/api/v1/rest/datastore/F-A0010-001](https://opendata.cwa.gov.tw/api/v1/rest/datastore/F-A0010-001)"

TARGET_REGIONS = ["北部地區", "中部地區", "南部地區", "東北部地區", "東部地區", "東南部地區"]

# ------------------------------------------
# HW10-1: 獲取天氣預報資料 (20%)
# ------------------------------------------
def fetch_weather_data(api_key: str) -> dict:
    """調用 CWA API 獲取一週農業天氣預報資料 (JSON)"""
    params = {
        "Authorization": api_key,
        "format": "JSON"
    }
    
    print("[HW10-1] 發送 API 請求至中央氣象署...")
    response = requests.get(API_URL, params=params, timeout=15)
    response.raise_for_status()
    data = response.json()
    
    # 觀察獲得的資料 (評分要求：使用 json.dumps 觀察)
    print("\n" + "="*50)
    print("[HW10-1] 觀察取得的 JSON 原始結構 (前 600 字元)：")
    print("="*50)
    print(json.dumps(data, indent=2, ensure_ascii=False)[:600] + "\n...")
    
    return data

# ------------------------------------------
# HW10-2: 分析資料，提取最高與最低氣溫 (20%)
# ------------------------------------------
def parse_temperature_data(raw_data: dict) -> list[dict]:
    """從 JSON 結構中提取指定六大地區的日期、最低氣溫與最高氣溫"""
    locations = raw_data.get("records", {}).get("locations", {}).get("location", [])
    extracted_records = []
    
    for loc in locations:
        region_name = loc.get("locationName")
        if region_name not in TARGET_REGIONS:
            continue
            
        weather_elements = {elem["elementName"]: elem for elem in loc.get("weatherElement", [])}
        mint_times = weather_elements.get("MinT", {}).get("time", [])
        maxt_times = weather_elements.get("MaxT", {}).get("time", [])
        
        # 對齊 MinT 與 MaxT 的日期時段
        temp_dict = {}
        for item in mint_times:
            # 格式：YYYY-MM-DD
            date_str = item["startTime"].split("T")[0]
            val = float(item["elementValue"][0]["value"])
            temp_dict.setdefault(date_str, {})["mint"] = val

        for item in maxt_times:
            date_str = item["startTime"].split("T")[0]
            val = float(item["elementValue"][0]["value"])
            temp_dict.setdefault(date_str, {})["maxt"] = val

        for date_str, temps in sorted(temp_dict.items()):
            if "mint" in temps and "maxt" in temps:
                extracted_records.append({
                    "regionName": region_name,
                    "dataDate": date_str,
                    "mint": temps["mint"],
                    "maxt": temps["maxt"]
                })
                
    # 觀察提取出來的資料 (評分要求：使用 json.dumps 觀察)
    print("\n" + "="*50)
    print("[HW10-2] 觀察提取後的氣溫結構 (範例前 3 筆)：")
    print("="*50)
    print(json.dumps(extracted_records[:3], indent=2, ensure_ascii=False))
    
    return extracted_records

# ------------------------------------------
# HW10-3: 儲存至 SQLite3 資料庫並驗證 (20%)
# ------------------------------------------
def init_and_save_to_db(records: list[dict], db_path: str = "data.db"):
    """建立資料表並存入氣溫資料，最後執行驗證查詢"""
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    # 建立 Table
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS TemperatureForecasts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            regionName TEXT NOT NULL,
            dataDate TEXT NOT NULL,
            mint REAL NOT NULL,
            maxt REAL NOT NULL,
            UNIQUE(regionName, dataDate) ON CONFLICT REPLACE
        )
    """)
    
    # 寫入資料
    insert_sql = """
        INSERT INTO TemperatureForecasts (regionName, dataDate, mint, maxt)
        VALUES (:regionName, :dataDate, :mint, :maxt)
    """
    cursor.executemany(insert_sql, records)
    conn.commit()
    print(f"\n[HW10-3] 成功將 {len(records)} 筆紀錄寫入 {db_path} 中的 TemperatureForecasts 表格。")
    
    # ---------------- 驗證查詢 ----------------
    print("\n" + "="*50)
    print("[HW10-3] 驗證 1：列出所有地區名稱")
    print("="*50)
    cursor.execute("SELECT DISTINCT regionName FROM TemperatureForecasts;")
    regions = cursor.fetchall()
    for reg in regions:
        print(f"- {reg[0]}")
        
    print("\n" + "="*50)
    print("[HW10-3] 驗證 2：列出中部地區的氣溫資料")
    print("="*50)
    cursor.execute("""
        SELECT id, regionName, dataDate, mint, maxt 
        FROM TemperatureForecasts 
        WHERE regionName = '中部地區'
        ORDER BY dataDate ASC;
    """)
    central_rows = cursor.fetchall()
    cols = ["id", "regionName", "dataDate", "mint", "maxt"]
    print(pd.DataFrame(central_rows, columns=cols).to_string(index=False))
    
    conn.close()

if __name__ == "__main__":
    if CWA_API_KEY == "YOUR_CWA_API_KEY_HERE":
        print("提示：請先將程式碼中的 CWA_API_KEY 替換為您的中央氣象署有效授權碼。")
    else:
        raw_json = fetch_weather_data(CWA_API_KEY)
        parsed_records = parse_temperature_data(raw_json)
        init_and_save_to_db(parsed_records)
```

### 2. `app.py` (HW10-4 完整整合應用)

```python
import sqlite3
import streamlit as st
import pandas as pd
import folium
from streamlit_folium import st_folium

# 設定頁面配置
st.set_page_config(page_title="台灣一週氣溫預報 Web App", layout="wide")

DB_PATH = "data.db"

# 六大分區參考座標
REGION_COORDS = {
    "北部地區": {"lat": 25.0330, "lon": 121.5654},
    "中部地區": {"lat": 24.1477, "lon": 120.6736},
    "南部地區": {"lat": 22.6273, "lon": 120.3014},
    "東北部地區": {"lat": 24.7021, "lon": 121.7377},
    "東部地區": {"lat": 23.9871, "lon": 121.6015},
    "東南部地區": {"lat": 22.7583, "lon": 121.1444}
}

def get_temp_color(temp: float) -> str:
    """依據平均氣溫傳回對應的色碼"""
    if temp < 20:
        return "blue"
    elif 20 <= temp < 25:
        return "green"
    elif 25 <= temp <= 30:
        return "orange"
    else:
        return "red"

@st.cache_data
def load_all_data():
    """從 SQLite 資料庫撈取所有資料"""
    try:
        conn = sqlite3.connect(DB_PATH)
        df = pd.read_sql_query("SELECT regionName, dataDate, mint, maxt FROM TemperatureForecasts", conn)
        conn.close()
        return df
    except Exception as e:
        st.error(f"讀取資料庫失敗，請確認是否已執行 etl_weather.py。錯誤：{e}")
        return pd.DataFrame()

df = load_all_data()

st.title("🌤️ 台灣一週氣溫預報 Web App")
st.caption("資料來源：中央氣象署 (CWA) 農業天氣預報 API (F-A0010-001) | 儲存核心：SQLite3")

if df.empty:
    st.warning("資料庫中無資料，請先執行 `python etl_weather.py` 獲取並建立資料庫。")
    st.stop()

# 左右版面配置
col_left, col_right = st.columns([1, 1], gap="large")

# ==========================================
# 左側：地圖視覺化 (Folium + 溫標顏色標記)
# ==========================================
with col_left:
    st.subheader("🗺️ 台灣氣溫地圖分佈")
    
    # 日期選擇器以動態呈現該日全台氣溫
    all_dates = sorted(df["dataDate"].unique())
    selected_date = st.selectbox("選擇地圖顯示日期：", options=all_dates)
    
    # 篩選當日數據
    date_df = df[df["dataDate"] == selected_date]
    
    # 建立 Folium 地圖中心
    m = folium.Map(location=[23.7, 120.9], zoom_start=7, tiles="CartoDB positron")
    
    for _, row in date_df.iterrows():
        reg = row["regionName"]
        if reg in REGION_COORDS:
            coords = REGION_COORDS[reg]
            avg_temp = (row["mint"] + row["maxt"]) / 2.0
            color = get_temp_color(avg_temp)
            
            popup_html = f"""
            <div style="font-family: Arial; min-width: 120px;">
                <h4>{reg}</h4>
                <b>日期:</b> {row['dataDate']}<br>
                <b>最低溫:</b> {row['mint']} °C<br>
                <b>最高溫:</b> {row['maxt']} °C<br>
                <b>平均溫:</b> {avg_temp:.1f} °C
            </div>
            """
            
            folium.CircleMarker(
                location=[coords["lat"], coords["lon"]],
                radius=14,
                color=color,
                fill=True,
                fill_color=color,
                fill_opacity=0.7,
                popup=folium.Popup(popup_html, max_width=300),
                tooltip=f"{reg} ({avg_temp:.1f} °C)"
            ).add_to(m)
            
    st_folium(m, width=500, height=520)
    st.caption("燈號說明：🔵 <20°C | 🟢 20-25°C | 🟠 25-30°C | 🔴 >30°C")

# ==========================================
# 右側：指定地區預報 (下拉選單 + 折線圖 + 表格)
# ==========================================
with col_right:
    st.subheader("📈 地區一週預報詳情")
    
    # HW10-4 下拉選單功能
    available_regions = sorted(df["regionName"].unique())
    selected_region = st.selectbox("請選擇查詢地區：", options=available_regions, index=0)
    
    # 篩選特定地區一週資料
    region_df = df[df["regionName"] == selected_region].sort_values("dataDate").reset_index(drop=True)
    
    # HW10-4 折線圖顯示
    st.markdown("**一週最低溫 vs 最高溫趨勢圖**")
    chart_data = region_df.set_index("dataDate")[["mint", "maxt"]]
    chart_data.columns = ["最低氣溫 (MinT)", "最高氣溫 (MaxT)"]
    st.line_chart(chart_data)
    
    # HW10-4 資料表格顯示
    st.markdown("**詳細氣溫數據表**")
    display_table = region_df[["dataDate", "mint", "maxt"]].copy()
    display_table.columns = ["預報日期", "最低氣溫 (°C)", "最高氣溫 (°C)"]
    st.dataframe(display_table, use_container_width=True, hide_index=True)
```

## 執行與驗收步驟

### 步驟 1：執行資料擷取與入庫
開啟終端機執行：
```bash
python etl_weather.py
```
* **驗證項目**：
  1. 終端機會依序印出 API 原始 JSON 前 600 字元與解析後的氣溫欄位。
  2. 專案根目錄自動建立 `data.db` 與 `TemperatureForecasts` 資料表。
  3. 自動執行 SQL 檢驗，列出六大分區清單與中部地區數據。

### 步驟 2：啟動 Streamlit 視覺化 App
```bash
streamlit run app.py
```
* **驗證項目**：
  1. 瀏覽器開啟 `http://localhost:8501`。
  2. 左側顯示全台地圖與日期篩選，各區以氣溫色彩圈標記。
  3. 右側支援下拉選單切換地區，即時更新一週折線圖與氣溫表格。
